def generate_world(seed=None):
    if seed is not None: random.seed(seed)
    num = random.randint(6, 9)
    regions = {}
    for i in range(num):
        row = i // 2; col = i % 2
        y = 100 + row * 130 + random.uniform(-20, 20)
        x = 150 + col * 200 + random.uniform(-30, 30)
        biome = random.choice(list(BIOMES.keys()))
        b = BIOMES[biome]
        name = random.choice(NP) + random.choice(NS)
        while name in regions: name = random.choice(NP) + random.choice(NS)
        enemies = random.sample(b["enemies"], k=min(len(b["enemies"]), random.randint(1,2)))
        regions[name] = {
            "x":x,"y":y,"r":random.randint(38,48),"biome":biome,
            "emoji":b["emoji"],"color":b["color"],"stroke":b["stroke"],
            "desc":random.choice(REGION_DESCS),
            "enemies":enemies,"danger":random.randint(1,5),
            "has_boss":False,"npc":None,
        }
    boss_region = random.choice(list(regions.keys()))
    regions[boss_region]["has_boss"] = True
    regions[boss_region]["enemies"] = []
    regions[boss_region]["biome"] = "ruínas"
    regions[boss_region]["stroke"] = "#ff5e5e"
    regions[boss_region]["emoji"] = "👁️"
    regions[boss_region]["desc"] = "O coração pulsante do mundo. Algo antigo dorme aqui."

    connections = []
    rl = list(regions.keys())
    for i in range(len(rl)-1): connections.append((rl[i], rl[i+1]))
    if len(rl) > 4:
        connections.append((rl[0], rl[2]))
        connections.append((rl[-1], rl[-3]))

    return {"regions":regions,"connections":connections,
            "boss_region":boss_region,"boss_name":random.choice(BOSS_NAMES)}

def new_world(seed=None):
    world = generate_world(seed)
    st.session_state.world = world
    st.session_state.world_seed = seed
    st.session_state.current_region = None
    st.session_state.visited = []
    st.session_state.boss_defeated = False
    st.session_state.game_complete = False
    st.session_state.quests_done = []
    st.session_state.view = "mapa"

    region_names = [r for r in world["regions"] if not world["regions"][r]["has_boss"]]
    npc_regions = random.sample(region_names, min(3, len(region_names)))
    used = set()
    for i, rname in enumerate(npc_regions):
        available = [q for q in QUEST_TEMPLATES if q["title"] not in used]
        if not available: break
        q = random.choice(available); used.add(q["title"])
        npc_name = random.choice(NPC_NAMES) + " " + random.choice(NS).capitalize()
        world["regions"][rname]["npc"] = {
            "name":npc_name,
            "title":random.choice(["Cartógrafo","Guardião","Escavador","Batedor","Sentinela"]),
            "dialog":random.choice([

                "O mundo mudou desde o último eclipse.",

                "Ajuda-me e eu te ajudarei.",

def add_log(text, cls="log-info"):
    st.session_state.log.append({"text":f"[T{st.session_state.turn}] {text}","cls":cls})

def add_item(name, qty=1):
    for it in st.session_state.inventory:
        if it["name"]==name: it["qty"]+=qty; return
    st.session_state.inventory.append({"name":name,"qty":qty})

def remove_item(name, qty=1):
    for it in st.session_state.inventory:
        if it["name"]==name: it["qty"]-=qty
    st.session_state.inventory = [i for i in st.session_state.inventory if i["qty"]>0]

def count_item(name):
    for it in st.session_state.inventory:
        if it["name"]==name: return it["qty"]
    return 0

def register_kill(enemy_name):
    st.session_state.kill_count[enemy_name] = st.session_state.kill_count.get(enemy_name,0)+1
    st.session_state.total_kills += 1

def add_xp(amount):
    st.session_state.p_xp += amount
    while st.session_state.p_xp >= st.session_state.p_xp_next:
        st.session_state.p_xp -= st.session_state.p_xp_next
        st.session_state.p_level += 1
        st.session_state.p_hp_max += 15
        st.session_state.p_hp = st.session_state.p_hp_max
        st.session_state.p_mp_max += 8
        st.session_state.p_mp = st.session_state.p_mp_max
        st.session_state.p_atk += 3
        st.session_state.p_def += 2
        st.session_state.p_mag += 2
        st.session_state.p_xp_next = int(100 * (1 + st.session_state.p_level * 0.5))
        add_log(f"🌟 LEVEL UP! Nível {st.session_state.p_level}!", "log-crit")

def check_quest_completion():
    for rname, rdata in st.session_state.world["regions"].items():
        npc = rdata.get("npc")
        if not npc: continue
        q = npc["quest"]
        if q["id"] in st.session_state.quests_done: continue
        req = q["requires"]
        done = False
        if "kill" in req: done = st.session_state.kill_count.get(req["kill"],0) >= req["count"]
        elif "item" in req: done = count_item(req["item"]) >= req["count"]
        if done:
            st.session_state.quests_done.append(q["id"])

def is_boss(): return st.session_state.in_boss

def get_enemy_stats():
    if is_boss():
        phase = BOSS_PHASES[st.session_state.boss_phase]
        return {"hp":st.session_state.boss_hp,"atk":phase["atk"],"def":phase["def"],
                "abilities":phase["abilities"],
                "name":f"{st.session_state.world['boss_name']} · {phase['name']}"}
    e = ENEMIES[st.session_state.enemy]
    return {**e,"name":st.session_state.enemy}

def player_attack():
    e = get_enemy_stats()
    base = st.session_state.p_atk
    w = st.session_state.equipped_weapon
    if w and "atk_bonus" in ITEMS.get(w,{}): base += ITEMS[w]["atk_bonus"]
    if st.session_state.player_buff_atk > 0:
        base = int(base*1.5); st.session_state.player_buff_atk -= 1
    roll = random.randint(1,20)
    crit = 10 + (20 if w == "Fragmento Caótico" else 0)
    if roll <= 2:
        add_log("💨 Você errou!", "log-sys")
        st.session_state.narration = narrate("miss")
        return
    dmg = max(1, base - e["def"] + random.randint(-3,3))
    if roll <= crit + 10:
        dmg *= 2; add_log(f"⚡ CRÍTICO! {dmg} dano!", "log-crit")
        st.session_state.narration = narrate("crit")
    else:
        add_log(f"⚔️ {dmg} de dano.", "log-dmg")
    if is_boss(): st.session_state.boss_hp -= dmg
    else: st.session_state.enemy_hp -= dmg

def player_defend():
    st.session_state.player_defending = True
    add_log("🛡 Defendendo (+50% DEF).", "log-info")

def player_cast_spell(spell_name):
    spell = SPELLS[spell_name]
    if st.session_state.p_mp < spell["cost"]:
        add_log("❌ Mana insuficiente!", "log-sys"); return False
    st.session_state.p_mp -= spell["cost"]
    e = get_enemy_stats()
    w = st.session_state.equipped_weapon
    mag = st.session_state.p_mag + (ITEMS.get(w,{}).get("mag_bonus",0) if w else 0)
    if spell["type"] == "damage":
        dmg = max(1, spell["power"] + mag - e["def"]//2)
        if is_boss(): st.session_state.boss_hp -= dmg
        else: st.session_state.enemy_hp -= dmg
        add_log(f"{spell['element']} {spell_name}: {dmg} dano!", "log-magic")
        if spell_name == "Raio Gélido" and random.random() < 0.3:
            st.session_state.enemy_frozen = True
            add_log("❄️ Congelado!", "log-magic")
    elif spell["type"] == "heal":
        st.session_state.p_hp = min(st.session_state.p_hp_max, st.session_state.p_hp+spell["power"])
        add_log(f"💚 +{spell['power']} HP.", "log-heal")
        st.session_state.narration = narrate("heal")
    elif spell["type"] == "buff":
        st.session_state.player_buff_def = 3
        add_log(f"🛡 +{spell['power']} DEF.", "log-magic")
    elif spell["type"] == "drain":
        dmg = spell["power"] + mag//2; heal = dmg//2
        if is_boss(): st.session_state.boss_hp -= dmg
        else: st.session_state.enemy_hp -= dmg
        st.session_state.p_hp = min(st.session_state.p_hp_max, st.session_state.p_hp+heal)
        add_log(f"🩸 {dmg} dano, +{heal} HP.", "log-magic")
    return True

def player_use_item(name):
    data = ITEMS.get(name,{})
    if data.get("heal"):
        st.session_state.p_hp = min(st.session_state.p_hp_max, st.session_state.p_hp+data["heal"])
        add_log(f"🧪 +{data['heal']} HP.", "log-heal")
    elif data.get("mana"):
        st.session_state.p_mp = min(st.session_state.p_mp_max, st.session_state.p_mp+data["mana"])
        add_log(f"🔮 +{data['mana']} MP.", "log-info")
    elif data.get("buff") == "atk":
        st.session_state.player_buff_atk = 3
        add_log(f"🔥 ATK +50% (3t).", "log-info")
    remove_item(name,1)

def enemy_turn():
    e = get_enemy_stats()
    if st.session_state.enemy_frozen:
        st.session_state.enemy_frozen = False
        add_log(f"❄️ Congelado, pula turno.", "log-sys"); return
    if random.random() < 0.3:
        ability = random.choice(e["abilities"])
        dmg = max(1, e["atk"] + random.randint(2,8) - st.session_state.p_def//2)
        add_log(f"💀 '{ability}' → {dmg}!", "log-dmg")
    else:
        dmg = max(1, e["atk"] + random.randint(-3,3) - st.session_state.p_def//2)
        add_log(f"💥 {dmg} dano.", "log-dmg")
    if st.session_state.player_defending:
        dmg //= 2; st.session_state.player_defending = False
        add_log(f"🛡 Reduzido para {dmg}.", "log-info")
    if st.session_state.player_buff_def > 0:
        dmg = max(1, dmg-10); st.session_state.player_buff_def -= 1
    st.session_state.p_hp -= dmg
    if st.session_state.p_hp > 0 and st.session_state.p_hp < st.session_state.p_hp_max * 0.3:
        st.session_state.narration = narrate("dmg")

def next_turn(): st.session_state.turn += 1

def check_combat_end():
    if is_boss():
        if st.session_state.boss_hp <= 0:
            if st.session_state.boss_phase < len(BOSS_PHASES)-1:
                st.session_state.boss_phase += 1
                phase = BOSS_PHASES[st.session_state.boss_phase]
                st.session_state.boss_hp = phase["hp"]
                st.session_state.boss_hp_max = phase["hp"]
                add_log(f"⚠️ {phase['name']}: {phase['dialogue']}", "log-crit")
                st.session_state.narration = narrate(f"boss_p{st.session_state.boss_phase+1}")
                st.session_state.enemy_frozen = False
                return False
            else:
                add_log("🏆 BOSS DERROTADO!", "log-crit")
                st.session_state.boss_defeated = True
                st.session_state.game_complete = True
                st.session_state.in_boss = False; st.session_state.in_combat = False
                st.session_state.view = "mapa"
                add_xp(500); st.session_state.total_gold += 1000
                st.session_state.narration = narrate("win")
                return True
        if st.session_state.p_hp <= 0:
            add_log("💀 Derrotado...", "log-dmg")
            st.session_state.narration = narrate("death")
            st.session_state.p_hp = st.session_state.p_hp_max//2
            st.session_state.in_boss = False; st.session_state.in_combat = False
            st.session_state.view = "mapa"
            return True
        return False
    if st.session_state.enemy_hp <= 0:
        e = ENEMIES[st.session_state.enemy]
        xp, gold = e["xp"], e["gold"]
        loot_msg = ""
        if random.random() < 0.35:
            loot = random.choice(["Poção de Restauração","Poção de Mana","Elixir de Fúria","Lâmina de Eco"])
            add_item(loot,1); loot_msg = f" | +{loot}"
        add_xp(xp); st.session_state.total_gold += gold
        register_kill(st.session_state.enemy)
        add_log(f"🏆 Vitória! +{xp}XP +{gold}💰{loot_msg}", "log-crit")
        st.session_state.in_combat = False; st.session_state.view = "mapa"
        st.session_state.narration = narrate("win")
        check_quest_completion()
        return True
    if st.session_state.p_hp <= 0:
        add_log("💀 Derrotado...", "log-dmg")
        st.session_state.narration = narrate("death")
        st.session_state.p_hp = max(1, st.session_state.p_hp_max//3)
        st.session_state.in_combat = False; st.session_state.view = "mapa"
        return True
    return False

def player_flee():
    roll = random.randint(1,20) + st.session_state.p_level
    if roll >= 12:
        add_log("🏃 Escapou!", "log-info")
        st.session_state.in_combat = False; st.session_state.view = "mapa"; return True
    else:
        if st.session_state.total_gold >= 50:
            lost = random.randint(20,80)
            st.session_state.total_gold = max(0, st.session_state.total_gold-lost)
            add_log(f"❌ Fuga falhou! -{lost}💰", "log-dmg")
        e = get_enemy_stats(); dmg = e["atk"] + random.randint(0,10)
        st.session_state.p_hp -= dmg
        add_log(f"💥 Ataque: {dmg} dano.", "log-dmg")
        return False

                <div class="card-sub" style="margin-top:8px;">O Boss Final · Nível 8+</div>
                </div>""", unsafe_allow_html=True)
                if st.session_state.boss_defeated:
                    st.success("✅ Já derrotado!")
                else:
                    if st.button("🚪 ENTRAR NO BOSS", type="primary", use_container_width=True):
                        start_boss(); st.rerun()

    # ----------------- NPCs -----------------
    elif view == "npc":
        st.markdown("### 🧙 Missões")
        any_npc = False
        for rname, d in st.session_state.world["regions"].items():
            npc = d.get("npc")
            if not npc: continue
            any_npc = True
            q = npc["quest"]
            done = q["id"] in st.session_state.quests_done
            req = q["requires"]
            prog = st.session_state.kill_count.get(req["kill"],0) if "kill" in req else count_item(req["item"])
            st.markdown(f"""<div class="card" style="border-left:4px solid {'#6dd47e' if done else '#c08aff'};">
            <div class="card-title">🧙 {npc['name']}</div>
            <div class="card-sub">{npc['title']} · {d['emoji']} {rname}</div>
            <div class="card-desc" style="font-style:italic;">"{npc['dialog']}"</div>
            <div class="card-desc" style="color:#8aa9ff;font-weight:700;margin-top:10px;">📜 {q['title']}</div>
            <div class="card-desc">{q['goal']}</div>
            <div class="card-sub">Progresso: {prog}/{req['count']}</div>
            <div class="card-sub">💰{q['reward_gold']} · ✨{q['reward_xp']} · 🎁{q['reward_item']}</div>
            </div>""", unsafe_allow_html=True)
        if not any_npc: st.info("Nenhum NPC neste mundo.")

    # ----------------- INVENTÁRIO -----------------
    elif view == "inv":
        st.markdown("### 🎒 Inventário")
        c1, c2 = st.columns(2)
        with c1: st.caption(f"⚔️ {st.session_state.equipped_weapon or 'Sem arma'}")
        with c2: st.caption(f"🛡 {st.session_state.equipped_armor or 'Sem armadura'}")
        if not st.session_state.inventory:
            st.info("Inventário vazio.")
        else:
            for i, item in enumerate(st.session_state.inventory):
                name = item["name"]; data = ITEMS.get(name,{})
                rarity = data.get("rarity","Comum"); effect = data.get("effect","-")
                st.markdown(f"""<div class="card rar-{rarity}">
                <div class="card-title">{name} <span class="card-sub">x{item['qty']}</span></div>
                <div class="card-desc">{effect}</div>
                <div class="card-sub">{rarity} · {data.get('type','item').upper()}</div>
                </div>""", unsafe_allow_html=True)
                if data.get("type") == "arma":
                    if st.button("Equipar arma", key=f"eqw_{i}", use_container_width=True):
                        st.session_state.equipped_weapon = name; st.rerun()
                elif data.get("type") == "armadura":
                    if st.button("Equipar armadura", key=f"eqa_{i}", use_container_width=True):
                        st.session_state.equipped_armor = name; st.rerun()
                elif data.get("type") == "consumível":
                    if st.button("Usar", key=f"us_{i}", use_container_width=True):
                        player_use_item(name); st.rerun()
                if st.button("Descartar", key=f"dr_{i}", use_container_width=True):
                    remove_item(name,1); st.rerun()

    # ----------------- LOJA -----------------
    elif view == "shop":
        st.markdown("### 🏪 Loja")
        for name, data in ITEMS.items():
            price = data["value"]
            st.markdown(f"""<div class="card rar-{data['rarity']}">
            <div class="card-title">{name}</div>
            <div class="card-desc">{data['effect']}</div>
            <div class="card-sub">{data['rarity']} · {price}💰</div>
            </div>""", unsafe_allow_html=True)
            if st.button(f"Comprar por {price}💰", key=f"b_{name}", use_container_width=True):
                if st.session_state.total_gold >= price:
                    st.session_state.total_gold -= price
                    add_item(name,1); check_quest_completion(); st.rerun()
                else: st.error("Ouro insuficiente.")

    # ----------------- STATUS -----------------
    elif view == "status":
        st.markdown("### 📊 Ficha")

        st.markdown(f"""<div class="card">
        <div class="card-title">Nível {st.session_state.p_level}</div>
        <div class="card-sub">XP {st.session_state.p_xp}/{st.session_state.p_xp_next}</div>
        </div>
        <div class="card">
        <div class="card-title">Atributos</div>
        <div class="card-desc">⚔️ ATK: {st.session_state.p_atk}</div>
        <div class="card-desc">🛡 DEF: {st.session_state.p_def}</div>
        <div class="card-desc">🔮 MAG: {st.session_state.p_mag}</div>
        </div>""", unsafe_allow_html=True)

        st.markdown("#### 📜 Magias")
        for sname, sinfo in SPELLS.items():
            st.markdown(f"""<div class="card" style="padding:10px;">
            <div class="card-title" style="font-size:.95rem;">{sinfo['element']} {sname}</div>
            <div class="card-sub">{sinfo['cost']} MP · {sinfo['desc']}</div>
            </div>""", unsafe_allow_html=True)

        st.markdown("#### ☠️ Abates")
        if st.session_state.kill_count:
            for k,v in st.session_state.kill_count.items(): st.caption(f"• {k}: {v}")
        else: st.caption("Nenhum ainda.")

        st.markdown("#### 🏆 Conquistas")
        ach = [
            ("Primeira Sangue", st.session_state.total_kills >= 1),
            ("Caçador", st.session_state.total_kills >= 5),
            ("Lendário", st.session_state.total_kills >= 15),
            ("Nível 5", st.session_state.p_level >= 5),
            ("Nível 10", st.session_state.p_level >= 10),
            ("Rico", st.session_state.total_gold >= 1000),
            ("Herói", st.session_state.boss_defeated),
        ]
        for name, done in ach:
            st.caption(f"{'✅' if done else '⬜'} {name}")

        st.markdown("---")
        st.markdown("#### 🌍 Novo Mundo")
        if st.button("🎲 Gerar Novo Mundo", type="primary", use_container_width=True):
            new_world(); st.rerun()
        seed_val = st.number_input("Seed (0 = aleatória)", value=0, step=1, min_value=0)
        if st.button("🎯 Gerar com Seed", use_container_width=True):
            new_world(seed=seed_val if seed_val != 0 else None); st.rerun()
